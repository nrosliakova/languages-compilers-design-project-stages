        self.expect_text(";", "expected ';' after bye")

        return ExitNode(token.line, token.column, value)

    def parse_if(self):
        token = self.expect_text("chat")
        condition = self.parse_expr()
        then_block = self.parse_block()
        else_block = None

        if self.at("nah"):
            self.eat()
            else_block = self.parse_block()

        return IfNode(
            token.line,
            token.column,
            condition,
            then_block,
            else_block,
        )

    def parse_block(self):
        opening = self.expect_text(
            self.OPEN, "expected the opening block marker"
        )
        statements = []
        exit_node = None

        while not self.at(self.CLOSE):
            if self.peek().kind == "eof":
                raise CompileError(
                    opening.line,
                    opening.column,
                    "block is never closed",
                )

            if exit_node is not None:
                self.error_here(
                    "statement after 'bye' in the same block"
                )

            if self.at("bye"):
                exit_node = self.parse_exit()
            else:
                statements.append(self.parse_statement())

        self.eat()

        if not statements and exit_node is None:
            raise CompileError(
                opening.line, opening.column, "empty block"
            )

        return BlockNode(
            opening.line,
            opening.column,
            statements,
            exit_node,
        )

'''

        token = self.peek()

'''

    if re.search(r"\b(minint|bigint|vibe|chat|bye)\b", text):
        raise SystemExit(f"{path} already migrated; no changes made.")

    output = []
    for line in text.splitlines():
        stripped = line.strip()

        if not stripped:
            output.append(line)
            continue

        if stripped in {"{", "}"}:
            indent = line[:len(line) - len(line.lstrip())]
            output.append(indent + (down if stripped == "{" else up))
            continue

        declaration = re.match(
            r"^(\s*)(i32|i64|bool)\s+(mut\s+)?", line
        )
        if declaration:
            indent, old_type, mutable = declaration.groups()
            line = (
                indent + types[old_type] + " "
                + (unlock if mutable else lock) + " "
                + line[declaration.end():]
            )

        line = re.sub(
            r"\b(if|else|exit|true|false)\b",
            lambda match: words[match.group()],
            line,
        )
        line = line.replace(":=", "=")

        if not re.match(r"^\s*(chat|nah)\b", line):
            line = line.rstrip() + ";"

        output.append(line)

    converted[path] = "\n".join(output) + "\n"

syntax_cases = [
    "else_without_if", "empty_block", "assignment_using_equality",
    "missing_if_block", "statement_after_block_exit", "unclosed_block",
]

for name in syntax_cases:
    for suffix in (".txt", ".expected"):
        source = Path("tests/ast_err") / (name + suffix)
        if not source.is_file():
            raise SystemExit(f"Missing verified test: {source}; no changes made.")

for path, text in converted.items():
    path.write_text(text, encoding="utf-8")

for name in syntax_cases:
    for suffix in (".txt", ".expected"):
        shutil.copyfile(
            Path("tests/ast_err") / (name + suffix),
            folder / (name + suffix),
        )

for name in ("single_equals", "if_brace_same_line"):
    for suffix in (".txt", ".expected"):
        (folder / (name + suffix)).unlink(missing_ok=True)

print(f"Migrated {len(converted)} semantic sources and 6 syntax cases.")
